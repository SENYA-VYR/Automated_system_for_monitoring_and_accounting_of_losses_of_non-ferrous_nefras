class SensorFilter {
private:
    float buffer[10];
    int index = 0;
    bool initialized = false;
public:
    float addValue(float value) {
        if (!initialized) {

            for(int i = 0; i < 10; i++) buffer[i] = value;
            initialized = true;
        }
        buffer[index] = value;
        index = (index + 1) % 10;
        return getAverage();
    }
    
    float getAverage() {
        float sum = 0;
        for(int i = 0; i < 10; i++) sum += buffer[i];
        return sum / 10;
    }
    
    float getStdDev() {
        float avg = getAverage();
        float variance = 0;
        for(int i = 0; i < 10; i++) {
            variance += pow(buffer[i] - avg, 2);
        }
        return sqrt(variance / 10);
    }
};

void setup() {
  Serial.begin(115200);

  preferences.begin("nephras-config", false);
  loadCredentials(); // Загрузка сохраненных учетных данных

  pinMode(LED_WARN, OUTPUT);
  pinMode(LED_ALARM, OUTPUT);

  initWatchdog();

  aes128.setKey(encryptionKey, 16);

  blinkLED(LED_OK, 3, 200);

  connectToWiFi();

  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("Wi-Fi недоступен. Попытка Ethernet...");
    backupSendData();
  }

  mqttClient.setServer(mqtt_server, mqtt_port);
  Serial.println("Датчиковый узел запущен с расширенными функциями.");
}

void loop() {
  unsigned long currentMillis = millis();

  resetWatchdog();

  for(int i = 0; i < 4; i++) {
    if(!checkSensorHealth(i)) {
      sensorHealth[i] = false;
      Serial.println("Ошибка датчика типа: " + String(i));
    }
  }

  if (!mqttClient.connected()) {
    connectToMQTT();
  }
  mqttClient.loop();

  if (currentMillis - previousMillis >= interval) {
    previousMillis = currentMillis;

    readSensors();
    checkForAnomalies();

    managePower();

    if (WiFi.status() == WL_CONNECTED || useEthernet || useGSM) {
      sendData();
      // Отправка данных из очереди, если есть
      while (!dataQueue.isEmpty() && (WiFi.status() == WL_CONNECTED || useEthernet || useGSM)) {
        String queuedData = dataQueue.pop();
        if (mqttClient.publish(mqtt_topic, queuedData.c_str())) {
          Serial.println("Данные из очереди отправлены.");
        }
      }
    } else {
      // Сохранение данных в очередь при отсутствии связи
      String payload = createPayload();
      if (dataQueue.count() < MAX_QUEUE_SIZE) {
        dataQueue.push(payload);
        Serial.println("Данные сохранены в очередь. Размер: " + String(dataQueue.count()));
      }
    }

    if (!systemOk) {
      digitalWrite(LED_OK, LOW);
      digitalWrite(LED_WARN, LOW);
      digitalWrite(LED_ALARM, HIGH);
      Serial.println("ТРЕВОГА: " + alarmMessage);
    } else {
      digitalWrite(LED_OK, HIGH);
      digitalWrite(LED_WARN, LOW);
      digitalWrite(LED_ALARM, LOW);
      Serial.println("Система в норме. Данные отправлены.");
    }
  }

  if (watchdogTriggered) {
    Serial.println("Watchdog сработал. Перезагрузка...");
    ESP.restart();
  }
}

String encryptData(String payload) {
  byte ciphertext[16];
  byte plaintext[16];

  payload.getBytes(plaintext, 16);

  aes128.encryptBlock(ciphertext, plaintext);

  String encrypted = base64::encode(ciphertext, 16);
  return encrypted;
}

String decryptData(String encryptedData) {
  byte ciphertext[16];
  byte decrypted[16];

  base64::decode(encryptedData).getBytes(ciphertext, 16);

  aes128.decryptBlock(decrypted, ciphertext);
  
  return String((char*)decrypted);
}

void initWatchdog() {
  watchdogTimer = timerBegin(0, 80, true);
  timerAttachInterrupt(watchdogTimer, &resetWatchdog, true);
  timerAlarmWrite(watchdogTimer, 1000000, true); 
  timerAlarmEnable(watchdogTimer);
}

void IRAM_ATTR resetWatchdog() {
  watchdogTriggered = false;
}

bool checkSensorHealth(int sensorType) {
  switch(sensorType) {
    case 0: // Датчик уровня
      float initialLevel = analogRead(LEVEL_SENSOR_ANALOG_PIN);
      delay(50);
      float currentLevel = analogRead(LEVEL_SENSOR_ANALOG_PIN);
      return abs(currentLevel - initialLevel) > 2;
    
    case 1: // Датчик давления
      return true;
    
    case 2: // Датчик температуры
      return true;
    
    case 3: // Датчик паров
      return true;
    
    default:
      return false;
  }
}

void readSensors() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  long duration = pulseIn(ECHO_PIN, HIGH);
  float distance_cm = duration * 0.034 / 2;
  float tankHeight_cm = 200.0;
  float rawLevel = ((tankHeight_cm - distance_cm) / tankHeight_cm) * 100.0;

  tankLevel = levelFilter.addValue(constrain(rawLevel, 0, 100));

  int pressureValue = analogRead(PRESSURE_SENSOR_PIN);
  float voltage = (pressureValue / 4095.0) * 3.3;
  float rawPressure = (voltage - 0.5) * (700.0 / 4.5);
  pressure = pressureFilter.addValue(rawPressure);

  int vaporValue = analogRead(VAPOR_SENSOR_PIN);
  float voltageVapor = (vaporValue / 4095.0) * 3.3;
  float rawVapor = voltageVapor * 1000;
  vaporConcentration = vaporFilter.addValue(rawVapor);

  Serial.println("=== Данные с фильтрацией ===");
  Serial.print("Уровень: "); Serial.print(tankLevel); 
  Serial.print(" (шум: ±"); Serial.print(levelFilter.getStdDev(), 2); Serial.println(")");
  // ... остальные датчики
}

String createPayload() {
  String payload = "{";
  payload += "\"node_id\":\"node01\",";
  payload += "\"timestamp\":" + String(millis()) + ",";
  payload += "\"level\":" + String(tankLevel, 2) + ",";
  payload += "\"pressure\":" + String(pressure, 2) + ",";
  payload += "\"temperature\":" + String(temperature, 2) + ",";
  payload += "\"vapor_concentration\":" + String(vaporConcentration, 2) + ",";
  payload += "\"sensor_health\":[" + String(sensorHealth[0]) + "," + String(sensorHealth[1]) + 
             "," + String(sensorHealth[2]) + "," + String(sensorHealth[3]) + "],";
  payload += "\"status\":" + String(systemOk ? "\"ok\"" : "\"alarm\"") + ",";
  payload += "\"alarm_message\":\"" + alarmMessage + "\"";
  payload += "}";

  String encryptedPayload = encryptData(payload);
  return encryptedPayload;
}

void sendData() {
  String payload = createPayload();
  
  if (mqttClient.publish(mqtt_topic, payload.c_str())) {
    Serial.println("Зашифрованные данные отправлены по MQTT.");
  } else {
    Serial.println("Ошибка отправки! Данные сохранены в очередь.");
    if (dataQueue.count() < MAX_QUEUE_SIZE) {
      dataQueue.push(payload);
    }
  }
}

void connectToWiFi() {
  Serial.print("Подключение к Wi-Fi");
  WiFi.begin(ssid, password);
  
  int attempts = 0;
  while (WiFi.status() != WL_CONNECTED && attempts < 20) {
    delay(1000);
    Serial.print(".");
    attempts++;
    blinkLED(LED_ALARM, 1, 500);
  }
  
  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\nПодключено к Wi-Fi!");
    Serial.print("IP-адрес: ");
    Serial.println(WiFi.localIP());
    saveCredentials(); // Сохранение успешных учетных данных
  } else {
    Serial.println("\nОшибка подключения к Wi-Fi");
  }
}
